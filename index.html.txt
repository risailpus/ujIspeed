    // Pemetaan Jari untuk Setiap Karakter
    const FINGER_MAP = {
      '1': 'Kelingking Kiri', 'q': 'Kelingking Kiri', 'a': 'Kelingking Kiri', 'z': 'Kelingking Kiri',
      '2': 'Jari Manis Kiri', 'w': 'Jari Manis Kiri', 's': 'Jari Manis Kiri', 'x': 'Jari Manis Kiri',
      '3': 'Jari Tengah Kiri', 'e': 'Jari Tengah Kiri', 'd': 'Jari Tengah Kiri', 'c': 'Jari Tengah Kiri',
      '4': 'Telunjuk Kiri', '5': 'Telunjuk Kiri', 'r': 'Telunjuk Kiri', 't': 'Telunjuk Kiri',
      'f': 'Telunjuk Kiri', 'g': 'Telunjuk Kiri', 'v': 'Telunjuk Kiri', 'b': 'Telunjuk Kiri',
      '6': 'Telunjuk Kanan', '7': 'Telunjuk Kanan', 'y': 'Telunjuk Kanan', 'u': 'Telunjuk Kanan',
      'h': 'Telunjuk Kanan', 'j': 'Telunjuk Kanan', 'n': 'Telunjuk Kanan', 'm': 'Telunjuk Kanan',
      '8': 'Jari Tengah Kanan', 'i': 'Jari Tengah Kanan', 'k': 'Jari Tengah Kanan', ',': 'Jari Tengah Kanan',
      '9': 'Jari Manis Kanan', 'o': 'Jari Manis Kanan', 'l': 'Jari Manis Kanan', '.': 'Jari Manis Kanan',
      '0': 'Kelingking Kanan', '-': 'Kelingking Kanan', '=': 'Kelingking Kanan', 'p': 'Kelingking Kanan',
      ';': 'Kelingking Kanan', '/': 'Kelingking Kanan',

    };

    // State Aplikasi
    let currentLevel = 1;
    let timeLimit = 60;
    let timeRemaining = 60;
    let elapsedSeconds = 0;
    let timerInterval = null;
    let isRunning = false;
    let isFinished = false;

    let targetText = "";
    let currentIndex = 0;
    let totalTyped = 0;
    let totalErrors = 0;

    // Elemen DOM
    const textDisplay = document.getElementById('text-display');
    const hiddenInput = document.getElementById('hidden-input');
    const typingBox = document.getElementById('typing-box');
    const focusOverlay = document.getElementById('focus-overlay');
    const fingerInstruction = document.getElementById('finger-instruction');

    const statWpm = document.getElementById('stat-wpm');
    const statAcc = document.getElementById('stat-acc');
    const statTime = document.getElementById('stat-time');
    const statTimeSub = document.getElementById('stat-time-sub');
    const statErr = document.getElementById('stat-err');

    const resultModal = document.getElementById('result-modal');
    const modalRank = document.getElementById('modal-rank');
    const modalFeedback = document.getElementById('modal-feedback');
    const resWpm = document.getElementById('res-wpm');
    const resAcc = document.getElementById('res-acc');
    const resErr = document.getElementById('res-err');

    // Pilih Teks Acak Sesuai Level
    function pickText(level) {
      const list = PRACTICE_TEXTS[level];
      const randomIndex = Math.floor(Math.random() * list.length);
      return list[randomIndex];
    }

    // Render Teks ke Layar
    function renderText() {
      targetText = pickText(currentLevel);
      textDisplay.innerHTML = '';
      targetText.split('').forEach((char, idx) => {
        const span = document.createElement('span');
        span.className = 'char';
        if (idx === 0) span.classList.add('current');
        span.textContent = char;
        textDisplay.appendChild(span);
      });
      updateFingerGuide();
    }

    // Perbarui Panduan Jari & Sorotan Tombol Keyboard
    function updateFingerGuide() {
      document.querySelectorAll('.key.next-target').forEach(k => k.classList.remove('next-target'));

      if (currentIndex >= targetText.length) {
        fingerInstruction.textContent = 'Selesai! Memuat teks berikutnya...';
        return;
      }

      const nextChar = targetText[currentIndex];
      const lowerChar = nextChar.toLowerCase();
      const fingerName = FINGER_MAP[lowerChar] || 'Jari Sesuai Posisi';
      const displayChar = nextChar === ' ' ? 'SPASI' : nextChar;

      fingerInstruction.textContent = `Jari Tugas: ${fingerName} (Tombol "${displayChar}")`;

      const keyEl = document.querySelector(`.key[data-key="${CSS.escape(lowerChar)}"]`);
      if (keyEl) {
        keyEl.classList.add('next-target');
      }
    }

    // Reset Sesi Latihan
    function resetSession() {
      clearInterval(timerInterval);
      timerInterval = null;
      isRunning = false;
      isFinished = false;
      currentIndex = 0;
      totalTyped = 0;
      totalErrors = 0;
      elapsedSeconds = 0;
      timeRemaining = timeLimit;

      hiddenInput.value = '';
      statWpm.textContent = '0';
      statAcc.textContent = '100%';
      statErr.textContent = '0';

      if (timeLimit === 0) {
        statTime.textContent = '0s';
        statTimeSub.textContent = 'Waktu Berjalan';
      } else {
        statTime.textContent = `${timeLimit}s`;
        statTimeSub.textContent = 'Hitung Mundur';
      }

      resultModal.classList.remove('show');
      renderText();
      focusInput();
    }

    // Mulai Timer
    function startTimer() {
      if (isRunning) return;
      isRunning = true;

      timerInterval = setInterval(() => {
        elapsedSeconds++;

        if (timeLimit > 0) {
          timeRemaining--;
          statTime.textContent = `${timeRemaining}s`;
          if (timeRemaining <= 0) {
            finishSession();
            return;
          }
        } else {
          statTime.textContent = `${elapsedSeconds}s`;
        }

        updateStats();
      }, 1000);
    }

    // Hitung WPM & Akurasi
    function updateStats() {
      const minutes = Math.max(elapsedSeconds / 60, 1 / 60);
      const correctChars = Math.max(totalTyped - totalErrors, 0);
      const wpm = Math.round((correctChars / 5) / minutes);
      const accuracy = totalTyped === 0 ? 100 : Math.max(0, Math.round(((totalTyped - totalErrors) / totalTyped) * 100));

      statWpm.textContent = wpm;
      statAcc.textContent = `${accuracy}%`;
      statErr.textContent = totalErrors;
    }

    // Akhiri Sesi & Tampilkan Hasil
    function finishSession() {
      clearInterval(timerInterval);
      isRunning = false;
      isFinished = true;
      updateStats();

      const finalWpm = parseInt(statWpm.textContent, 10);
      const finalAcc = statAcc.textContent;

      resWpm.textContent = finalWpm;
      resAcc.textContent = finalAcc;
      resErr.textContent = totalErrors;

      if (finalWpm >= 50) {
        modalRank.textContent = 'Predikat: Master 10 Jari 🏆';
        modalFeedback.textContent = 'Luar biasa! Kecepatan mengetikmu sudah melampaui rata-rata dan siap untuk kebutuhan akademik maupun lomba.';
      } else if (finalWpm >= 30) {
        modalRank.textContent = 'Predikat: Pelajar Tangkas ⭐';
        modalFeedback.textContent = 'Bagus sekali! Posisi jarimu sudah semakin terbiasa. Tingkatkan sedikit lagi agar menembus 50 WPM.';
      } else {
        modalRank.textContent = 'Predikat: Pemula Tekun 🌱';
        modalFeedback.textContent = 'Awal yang baik! Pertahankan posisi jari di Home Row (ASDF - JKL;) dan hindari melihat keyboard saat mengetik.';
      }

      resultModal.classList.add('show');
    }

    // Tangani Input Ketikan
    hiddenInput.addEventListener('input', () => {
      if (isFinished) return;

      const typedValue = hiddenInput.value;
      if (!typedValue) return;

      if (!isRunning) {
        startTimer();
      }

      const typedChar = typedValue[typedValue.length - 1];
      hiddenInput.value = '';

      const chars = textDisplay.querySelectorAll('.char');
      const expectedChar = targetText[currentIndex];

      totalTyped++;

      if (typedChar === expectedChar) {
        chars[currentIndex].classList.remove('current', 'incorrect');
        chars[currentIndex].classList.add('correct');
        currentIndex++;

        if (currentIndex < targetText.length) {
          chars[currentIndex].classList.add('current');
          updateFingerGuide();
        } else {
          // Jika mode waktu bebas selesai 1 kalimat, akhiri sesi; jika mode timer, muat kalimat baru
          if (timeLimit === 0) {
            finishSession();
            return;
          } else {
            currentIndex = 0;
            renderText();
          }
        }
      } else {
        totalErrors++;
        chars[currentIndex].classList.add('incorrect');
      }

      updateStats();
    });

    // Efek Visual Penekanan Tombol Fisik
    document.addEventListener('keydown', (e) => {
      if (isFinished) return;

      // Fokus otomatis jika user langsung mengetik
      if (document.activeElement !== hiddenInput && !e.ctrlKey && !e.metaKey && !e.altKey) {
        focusInput();
      }

      const keyLower = e.key.length === 1 ? e.key.toLowerCase() : e.code;
      const keyEl = document.querySelector(`.key[data-key="${CSS.escape(keyLower)}"]`) ||
                    document.querySelector(`.key[data-key="${CSS.escape(e.key)}"]`);

      if (keyEl) {
        keyEl.classList.add('pressed');
      }
    });

    document.addEventListener('keyup', (e) => {
      const keyLower = e.key.length === 1 ? e.key.toLowerCase() : e.code;
      const keyEl = document.querySelector(`.key[data-key="${CSS.escape(keyLower)}"]`) ||
                    document.querySelector(`.key[data-key="${CSS.escape(e.key)}"]`);

      if (keyEl) {
        keyEl.classList.remove('pressed');
      }
    });

    // Fokus Management
    function focusInput() {
      hiddenInput.focus();
      typingBox.classList.add('focused');
      focusOverlay.classList.add('hidden');
    }

    typingBox.addEventListener('click', focusInput);
    focusOverlay.addEventListener('click', focusInput);

    hiddenInput.addEventListener('blur', () => {
      typingBox.classList.remove('focused');
      if (!isFinished) {
        focusOverlay.classList.remove('hidden');
      }
    });

    // Tombol Level
    document.querySelectorAll('[data-level]').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('[data-level]').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        currentLevel = parseInt(btn.dataset.level, 10);
        resetSession();
      });
    });

    // Tombol Waktu
    document.querySelectorAll('[data-time]').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('[data-time]').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        timeLimit = parseInt(btn.dataset.time, 10);
        resetSession();
      });
    });

    document.getElementById('btn-reset').addEventListener('click', resetSession);
    document.getElementById('btn-modal-again').addEventListener('click', resetSession);

    // Inisialisasi Awal
    resetSession();
  </script>
